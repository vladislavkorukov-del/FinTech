cmake_minimum_required(VERSION 3.22.1)
project(finora_local_llm LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# NDK r28 defaults to flexible page sizes. Enforce this on both packaged DSOs.
add_link_options("-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
if(NOT DEFINED FINORA_MNN_SOURCE)
    get_filename_component(FINORA_PROJECT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
    set(FINORA_MNN_SOURCE "${FINORA_PROJECT_ROOT}/.native/mnn-src")
endif()
if(NOT EXISTS "${FINORA_MNN_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Pinned MNN source missing. Run bash tool/local_llm/prepare_mnn.sh before building.")
endif()
execute_process(COMMAND git -C "${FINORA_MNN_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE MNN_ACTUAL_REV OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT MNN_ACTUAL_REV STREQUAL "b0239771b07b317a258fbc384683bbd540741406")
    message(FATAL_ERROR "MNN source revision differs from the verified model/runtime manifest")
endif()
set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM_OMNI OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(MNN_SEP_BUILD OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_QUANTOOLS OFF CACHE BOOL "" FORCE)
set(MNN_LLM_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "" FORCE)
set(LLM_SUPPORT_HTTP_RESOURCE OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL OFF CACHE BOOL "" FORCE)
set(MNN_VULKAN OFF CACHE BOOL "" FORCE)
set(MNN_SME2 OFF CACHE BOOL "" FORCE)
set(MNN_ARM82 ON CACHE BOOL "" FORCE)
set(LLM_LOG_TO_STRING ON CACHE BOOL "" FORCE)
add_subdirectory("${FINORA_MNN_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/mnn")
add_library(finora_local_llm SHARED local_llm/session.cpp local_llm/jni.cpp)
target_include_directories(finora_local_llm PRIVATE
    "${FINORA_MNN_SOURCE}/include"
    "${FINORA_MNN_SOURCE}/transformers/llm/engine/include")
target_link_libraries(finora_local_llm PRIVATE MNN android log)
